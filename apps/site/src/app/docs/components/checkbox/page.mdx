# Checkbox

A three-state checkbox. Composes well with the native `<label>` element and supports the **indeterminate** state needed for hierarchical lists (e.g. "select all" with mixed selections).

Built on **Radix Checkbox** so ARIA roles, keyboard activation (Space), focus management, and the indeterminate state come for free. Pixel-perfect port of Figma node `2331:76`.

## Installation

<InstallCommand name="checkbox" />

Or via npm:

```bash
pnpm add @petpooja/ui
```

## Usage

```tsx
import { Checkbox } from '@petpooja/ui';

export function Demo() {
  return (
    <label className="flex items-center gap-3">
      <Checkbox />
      <span>I agree to the terms</span>
    </label>
  );
}
```

## Controlled

Drive the checked value from React state with `checked` + `onCheckedChange`.

<ComponentPreview
  previewClassName="!min-h-[140px] !p-8"
  preview={<CheckboxControlledDemo />}
  code={`'use client';
import { useState } from 'react';
import { Checkbox } from '@petpooja/ui';

export function Demo() {
  const [agreed, setAgreed] = useState(false);
  return (
    <label className="flex items-center gap-3">
      <Checkbox
        checked={agreed}
        onCheckedChange={(v) => setAgreed(v === true)}
      />
      <span>I agree to the terms and conditions</span>
    </label>
  );
}`}
/>

## Indeterminate (select-all pattern)

The most common use of the indeterminate state: a parent checkbox whose value reflects "all", "none", or "some" of its children.

<ComponentPreview
  previewClassName="!min-h-[240px] !p-8"
  preview={<CheckboxIndeterminateDemo />}
  code={`'use client';
import { useState } from 'react';
import type { CheckedState } from '@radix-ui/react-checkbox';
import { Checkbox } from '@petpooja/ui';

const ITEMS = ['Eggs', 'Milk', 'Bread'];

export function Demo() {
  const [selected, setSelected] = useState({
    Eggs: true,
    Milk: false,
    Bread: false,
  });

  const checkedCount = ITEMS.filter((i) => selected[i]).length;
  const parent: CheckedState =
    checkedCount === 0
      ? false
      : checkedCount === ITEMS.length
      ? true
      : 'indeterminate';

  const toggleAll = (next: CheckedState) => {
    const value = next === true;
    setSelected({ Eggs: value, Milk: value, Bread: value });
  };

  return (
    <div className="flex flex-col gap-3">
      <label className="flex items-center gap-3">
        <Checkbox checked={parent} onCheckedChange={toggleAll} />
        <span>Select all</span>
      </label>
      {ITEMS.map((item) => (
        <label key={item} className="flex items-center gap-3">
          <Checkbox
            checked={selected[item]}
            onCheckedChange={(v) =>
              setSelected((prev) => ({ ...prev, [item]: v === true }))
            }
          />
          <span>{item}</span>
        </label>
      ))}
    </div>
  );
}`}
/>
